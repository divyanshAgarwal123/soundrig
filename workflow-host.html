    Connect your own game
    import {createPlayer} from './component.mjs';
import {createGameAudio, attachPhaserScene} from './game-adapter.mjs';
// Load bindings.json, then connect your actual raw fields:
const sound = createGameAudio({player: createPlayer(), bindings,
  initial: {vx:0, vy:0, onGround:true, gameOver:false}});
await sound.start(); // Inside your Enable audio click.
sound.state({vx, vy, onGround, gameOver});
sound.event('collected'); // A real game callback, not inferred.
// Phaser: attachPhaserScene(scene, {player, bindings, readState}).
// It owns its listeners and stops on pause/sleep/shutdown.
// For plain JS call sound.interrupt() and sound.dispose() yourself.
    Own volume and mute settings apply to this stream. It does not inherit Phaser's sound manager settings. Check expected events against your game outcomes; a matching adapter/recording can share a wrong mapping.
